BASIC WORKFLOW COMMANDS
1. Status & Information

# Check repository status
git status
git status -s                 # Short format
git status --short
git status --porcelain        # Machine-readable format

# Show changes
git diff                      # Unstaged changes
git diff --staged             # Staged changes (--cached works too)
git diff HEAD                 # All changes (staged + unstaged)
git diff <commit1> <commit2>  # Compare two commits
git diff <branch1> <branch2>  # Compare two branches

# View commit history
git log
git log --oneline             # Compact view
git log --graph               # Show branch structure
git log --all --graph         # All branches with graph
git log -p                    # Show patch (changes)
git log --stat                # Show statistics
git log --since="2 weeks ago"
git log --until="2024-01-01"
git log --author="name"
git log --grep="search-term"
git log -S "function-name"    # Search by code changes
git log --no-merges           # Exclude merge commits

Staging Changes
bash
# Add files to staging area
git add <file-name>           # Add specific file
git add .                     # Add all files in current directory
git add -A                    # Add all files (including deletions)
git add --all
git add *.js                  # Add all JS files
git add src/                  # Add all files in src directory
git add -p                    # Interactive add (patch mode)
git add -u                    # Add modified/deleted files only (not new)
git add -N <file>             # Add but don't track (intent to add)


3. Committing Changes
bash
# Create commit
git commit -m "Commit message"
git commit -m "Title" -m "Description"  # Multi-line message

# Commit with skipping staging area
git commit -am "Message"      # Add all tracked files and commit

# Amend last commit
git commit --amend            # Edit last commit message
git commit --amend -m "New message"
git commit --amend --no-edit  # Add changes to last commit without editing message

# Empty commit (for CI triggers)
git commit --allow-empty -m "Trigger deployment"

# Sign commit with GPG
git commit -S -m "Signed commit"

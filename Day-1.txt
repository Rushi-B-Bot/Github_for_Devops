# Push to remote
git push                      # Push current branch to upstream
git push origin <branch>      # Push specific branch
git push -u origin <branch>   # Push and set upstream
git push --all                # Push all branches
git push --tags               # Push all tags
git push --force              # Force push (DANGEROUS!)
git push --force-with-lease   # Safer force push
git push origin --delete <branch>  # Delete remote branch

# Pull from remote
git pull                      # Fetch + merge
git pull origin <branch>
git pull --rebase             # Fetch + rebase instead of merge
git pull --ff-only            # Only fast-forward
git pull --no-commit          # Pull but don't auto-commit

# Fetch updates
git fetch                     # Get updates without merging
git fetch origin
git fetch --all               # Fetch from all remotes
git fetch --prune             # Remove deleted remote branches



 Merge branches
git merge <branch-name>
git merge --no-ff <branch>    # Always create merge commit
git merge --squash <branch>   # Squash all commits into one
git merge --abort             # Abort merge in conflict

# Rebase
git rebase <branch>           # Rebase current branch onto another
git rebase --interactive HEAD~3  # Interactive rebase last 3 commits
git rebase -i <commit>        # Interactive from specific commit
git rebase --continue         # Continue after resolving conflicts
git rebase --skip             # Skip current commit
git rebase --abort            # Abort rebase

# Cherry-pick
git cherry-pick <commit>      # Apply specific commit
git cherry-pick <commit1> <commit2> <commit3>
git cherry-pick --no-commit <commit>  # Apply without committing
git cherry-pick -x <commit>   # Add original commit hash to message 


# Update local from remote
git fetch && git merge origin/main
git pull --rebase origin main

# Push to multiple remotes
# Add multiple push URLs:
git remote set-url --add --push origin <url1>
git remote set-url --add --push origin <url2>
# Then just: git push

# Mirror repository
git clone --mirror <url>
git push --mirror

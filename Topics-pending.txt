# List tags
git tag                       # List all tags
git tag -l "v1.*"             # Filter tags with pattern
git tag -n                    # Show tag messages

# Create tags
git tag <tag-name>            # Lightweight tag
git tag -a <tag-name> -m "Message"  # Annotated tag
git tag -a v1.2 <commit-hash> # Tag specific commit
git tag -s <tag-name> -m "Message"  # Signed tag

# Delete tags
git tag -d <tag-name>         # Delete local tag
git push origin --delete <tag> # Delete remote tag
git push origin :refs/tags/<tag>  # Alternative delete

# Push tags
git push origin <tag-name>    # Push specific tag
git push origin --tags        # Push all tags

# Checkout tags
git checkout <tag-name>       # Creates detached HEAD
git checkout -b <branch> <tag> # Create branch from tag  

# Binary search for bugs
git bisect start
git bisect bad                # Current commit is bad
git bisect good <commit>      # Mark known good commit
# Git checks out midpoint, you test...
git bisect good               # If current is good
git bisect bad                # If current is bad
git bisect reset              # Finish

# Find lost commits
git reflog                    # Show reference log
git fsck --lost-found         # Find dangling commits

# Find large files
git rev-list --objects --all | \
  git cat-file --batch-check='%(objecttype) %(objectname) %(objectsize) %(rest)' | \
  sed -n 's/^blob //p' | \
  sort --numeric-sort --key=2 | \
  tail -10 | \
  cut -c 1-12,41-



# Add submodule
git submodule add <repository-url> <path>

# Clone with submodules
git clone --recursive <url>
# OR after clone:
git submodule init
git submodule update

# Update submodules
git submodule update --remote  # Update to latest
git submodule update --init --recursive

# Remove submodule
git submodule deinit <path>
git rm <path>
rm -rf .git/modules/<path>




# Optimize repository
git gc                        # Garbage collection
git gc --aggressive           # More aggressive cleanup
git gc --auto                 # Run auto cleanup

# Verify repository integrity
git fsck                      # Check connectivity
git fsck --full

# Repack objects
git repack -a -d --depth=250 --window=250

# Show repository size
git count-objects -vH

# Remove large files from history (BFG or git-filter-repo)
# Install: brew install git-filter-repo
git filter-repo --strip-blobs-bigger-than 100M
